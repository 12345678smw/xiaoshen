    // 用 setTimeout 模拟每一步的耗时
function sleep(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

async function cookDumplings() {
  // ① 买原料：线上、线下同时进行，谁先完成用谁的
  const source = await Promise.race([
    onlineOrder(6000).then(() => '线上'),
    offlineBuy(4500).then(() => '线下')
  ]);

  // ② 取原料
  await takeIngredients(1500);

  // ③ 和面、和馅同时进行，两者都完成才继续
  await Promise.all([kneadDough(3000), mixFilling(2500)]);

  // ④ 包饺子 → 煮饺子 → 吃饺子
  await wrapDumplings(4000);
  await boilDumplings(5000);
  await eatDumplings(2000);
}
